rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /{document=**} {
      // Parents: can see and change everything. Use lowercase emails.
      allow read, write: if request.auth != null
        && request.auth.token.email in ['parent1@example.com', 'parent2@example.com'];

      // Optional: let your daughters SEE their balances (no changes).
      // Add their emails here and create accounts for them too.
      allow read: if request.auth != null
        && request.auth.token.email in ['daughter1@example.com', 'daughter2@example.com'];
    }
  }
}
